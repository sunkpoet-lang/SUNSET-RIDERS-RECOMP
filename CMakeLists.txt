cmake_minimum_required(VERSION 3.24)
project(SunsetRidersResearch VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "This initial source export supports Windows UCRT64 only")
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Only the documented UCRT64 GCC toolchain is currently validated")
endif()
# Assertions are part of the runtime validation contract, including Release builds.
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-UNDEBUG>)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
find_package(SDL3 CONFIG REQUIRED)
include(CTest)
set(R ${CMAKE_CURRENT_SOURCE_DIR}/third_party/reference)
set(M ${R}/mame/m68000)
set(G ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(S ${CMAKE_CURRENT_SOURCE_DIR}/src)
foreach(PART header:m68000-head.h decode:m68000-decode.cpp sdf:m68000-sdf.cpp sdp:m68000-sdp.cpp)
  string(REPLACE ":" ";" PAIR "${PART}")
  list(GET PAIR 0 MODE)
  list(GET PAIR 1 NAME)
  add_custom_command(OUTPUT ${G}/cpu/m68000/${NAME}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${G}/cpu/m68000
    COMMAND Python3::Interpreter ${M}/m68000gen.py ${MODE} ${M}/m68000.lst ${G}/cpu/m68000/${NAME}
    DEPENDS ${M}/m68000gen.py ${M}/m68000.lst VERBATIM)
  list(APPEND MAIN_GENERATED ${G}/cpu/m68000/${NAME})
endforeach()
add_library(maincpu STATIC ${M}/m68000.cpp ${M}/m68kdasm.cpp ${R}/m68000_spaces.cpp ${MAIN_GENERATED})
target_include_directories(maincpu PUBLIC ${R}/stub ${M} ${G}/cpu/m68000)
set(Z ${CMAKE_CURRENT_SOURCE_DIR}/third_party/z80)
add_custom_command(OUTPUT ${G}/cpu/z80/z80.hxx
  COMMAND ${CMAKE_COMMAND} -E make_directory ${G}/cpu/z80
  COMMAND Python3::Interpreter ${Z}/z80make.py z80 ${Z}/z80.lst ${G}/cpu/z80/z80.hxx
  DEPENDS ${Z}/z80make.py ${Z}/z80.lst VERBATIM)
add_library(soundcpu OBJECT ${S}/z80_core.cpp ${S}/window_sound.cpp ${G}/cpu/z80/z80.hxx)
target_include_directories(soundcpu PRIVATE ${S} host/z80 ${R}/sound ${Z} ${G} ${R}/ymfm)
add_library(pcm OBJECT ${S}/window_pcm.cpp)
target_include_directories(pcm PRIVATE ${S} host/pcm third_party/devices ${R}/sound ${R}/ymfm)
add_library(video OBJECT ${S}/window_video.cpp)
target_include_directories(video PRIVATE ${S} host/video third_party/devices ${R}/mame ${R}/mame/util ${R}/mame/emu ${R}/mame/konami)
add_library(ymfm STATIC ${R}/ymfm/ymfm_opm.cpp)
target_include_directories(ymfm PUBLIC ${R}/ymfm)
add_library(shared_menu STATIC ${S}/shared/menu.cpp ${S}/shared/menubar.cpp)
target_include_directories(shared_menu PUBLIC ${S}/shared)
# A source/compiler/profile fingerprint binds private checkpoint codecs to this build.
file(GLOB_RECURSE ID_FILES CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} src/* host/* third_party/*)
set(IDENTITY "${CMAKE_CXX_COMPILER_ID}:${CMAKE_CXX_COMPILER_VERSION}:r51")
foreach(F IN LISTS ID_FILES)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${F}")
  file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/${F} HASH)
  string(APPEND IDENTITY "${F}:${HASH}\n")
endforeach()
string(SHA256 ENGINE_ID "${IDENTITY}")
configure_file(assets/icon.rc.in ${CMAKE_CURRENT_BINARY_DIR}/icon.rc @ONLY)
add_executable(Sunset-Riders-Lab ${S}/window_machine.cpp ${CMAKE_CURRENT_BINARY_DIR}/icon.rc)
target_include_directories(Sunset-Riders-Lab PRIVATE ${S} ${S}/shared)
target_compile_definitions(Sunset-Riders-Lab PRIVATE SUNSET_STATE_ENGINE_ID="${ENGINE_ID}")
target_link_libraries(Sunset-Riders-Lab PRIVATE maincpu soundcpu pcm video ymfm shared_menu SDL3::SDL3 bcrypt psapi)
# Online play: rollback for up to four players (src/net, the seat match) on GekkoNet (BSD-2-Clause), its library
# only, static, without asio (the seat layer has its own UDP transport), at a pinned release.
include(FetchContent)
set(NO_ASIO_BUILD ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(gekkonet
  URL https://github.com/HeatXD/GekkoNet/archive/refs/tags/v20261024222910-9da2363.tar.gz
  URL_HASH SHA256=a9001dec8698120fe45b314527d181c403ee92e3a30a03d765d3e792b8c74e1c
  SOURCE_SUBDIR GekkoLib
  DOWNLOAD_EXTRACT_TIMESTAMP ON)
FetchContent_MakeAvailable(gekkonet)
find_package(ZLIB REQUIRED)
# The online window, and the headless peer the online checks run (tools/online_check.py)
add_executable(Sunset-Riders-Online WIN32 ${S}/sunset_online.cpp ${CMAKE_CURRENT_BINARY_DIR}/icon.rc)
add_executable(sunset-seat-peer ${S}/sunset_seat_peer.cpp)
foreach(T Sunset-Riders-Online sunset-seat-peer)
  target_include_directories(${T} PRIVATE ${S} ${S}/shared ${S}/net)
  target_compile_definitions(${T} PRIVATE SUNSET_STATE_ENGINE_ID="${ENGINE_ID}")
  target_link_libraries(${T} PRIVATE maincpu soundcpu pcm video ymfm shared_menu SDL3::SDL3 GekkoNet ZLIB::ZLIB bcrypt psapi ws2_32)
endforeach()
add_custom_command(TARGET Sunset-Riders-Online POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/assets/title-icon.bmp $<TARGET_FILE_DIR:Sunset-Riders-Online>
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:Sunset-Riders-Online> VERBATIM)
foreach(T maincpu soundcpu pcm video ymfm shared_menu Sunset-Riders-Lab Sunset-Riders-Online sunset-seat-peer)
  target_compile_definitions(${T} PRIVATE _GLIBCXX_ASSERTIONS)
endforeach()
add_custom_command(TARGET Sunset-Riders-Lab POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/assets/title-icon.bmp $<TARGET_FILE_DIR:Sunset-Riders-Lab>
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:Sunset-Riders-Lab> VERBATIM)
if(BUILD_TESTING)
  foreach(N menu bar_model assists turbo menu_input session_profile)
    add_executable(test_${N} ${S}/sunset_${N}_test.cpp)
    target_include_directories(test_${N} PRIVATE ${S} ${S}/shared)
    target_link_libraries(test_${N} PRIVATE shared_menu)
    add_test(NAME ${N} COMMAND test_${N})
  endforeach()
  add_test(NAME core_self_test COMMAND Sunset-Riders-Lab --self-test)
  add_test(NAME embedded_icon COMMAND Sunset-Riders-Lab --resource-test)
  add_test(NAME rom_loader COMMAND Python3::Interpreter -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tools -p test_rom_loader.py)
endif()
